独立 WebGPU 体积 ​

本节由codex施工

createVolumeScene 直接消费 IVolumeSource，按视图需要调度分块，不要求业务侧先展开完整三维数组。示例使用与独立体积成品相同的物理比例、初始相机和 512 MiB 分块预算；轴切片由按钮按需创建。

运行示例 ​

示例默认加载真实 ZIP，可直接调整体积参数并操作 X、Y、Z 轴切片。

新窗口打开完整示例

<script setup lang="ts">
// 使用 volume-scene 创建可交互的真实 WebGPU 体积与路线剖面。
import { onBeforeUnmount, onMounted, reactive, ref } from 'vue'
import { createVolumeScene } from 'm4-w-fast/volume-scene'
import type {
    IVolumeCameraState,
    IVolumeLockedSlice,
    IVolumeRouteProfile,
    IVolumeScene,
    IVolumeSource,
    VolumeQualityPreset,
    VolumeRenderMode,
    VolumeSliceAxis,
} from 'm4-w-fast/volume-scene'
import { createVolumeSection, createVolumeSectionRouteProfile } from 'm4-w-fast/volume-section'
import {
    createPhysicalVolumeCamera,
    createPhysicalVolumeLayout,
    defaultSectionRoute,
    legendBackground,
    loadDemoVolume,
    loadVolumeFile,
    volumeColorStops,
    volumeOpacityStops,
} from './verticalSectionDemoData'

const canvas = ref<HTMLCanvasElement>()
const overlay = ref<HTMLDivElement>()
const fileInput = ref<HTMLInputElement>()
const toolbarCollapsed = ref(false)
const quality = ref<VolumeQualityPreset>('auto')
const mode = ref<VolumeRenderMode>('composite')
const projection = ref<'perspective' | 'orthographic'>('perspective')
const sliceTool = ref<VolumeSliceAxis | 'auto' | 'none'>('auto')
const timeIndex = ref(0)
const times = ref<readonly string[]>([])
const opacity = ref(0.3)
const density = ref(100)
const verticalScale = ref(0)
const colorMode = ref<'step' | 'smooth'>('step')
const memoryLimitMb = ref(512)
const maxLockedSlices = ref(12)
const routeStart = reactive({ ...defaultSectionRoute[0] })
const routeEnd = reactive({ ...defaultSectionRoute[1] })
const routeSampleCount = ref(400)
const routeSampleMode = ref<'cell' | 'interpolate'>('interpolate')
const status = ref('正在加载真实 ZIP')
const performanceText = ref('等待 WebGPU 场景')
const statisticsText = ref('尚未初始化')
const metadataText = ref('尚无数据')
const profileText = ref('尚未生成')
const slicesText = ref('[]')
let scene: IVolumeScene | null = null
let source: IVolumeSource | null = null
let timer = 0
let resizeObserver: ResizeObserver | null = null
let profileRevision = 0
let currentProfile: IVolumeRouteProfile | null = null
const axisPositions = reactive({ x: 101.2, y: 6000, z: 36.8 })
const axisSteps = reactive({ x: 0.01, y: 1, z: 0.01 })
const lockedSliceIds = reactive<Record<VolumeSliceAxis, string | null>>({
    x: null,
    y: null,
    z: null,
})
const clip = reactive({
    minX: 0,
    maxX: 0,
    minY: 0,
    maxY: 0,
    minZ: 0,
    maxZ: 0,
})

const updateScene = () => {
    scene?.update({
        mode: mode.value,
        quality: quality.value,
        opacity: opacity.value,
        densityScale: density.value,
        verticalScale: verticalScale.value + 1,
        colorMode: colorMode.value,
    })
}

const toggleToolbar = () => {
    toolbarCollapsed.value = !toolbarCollapsed.value
    requestAnimationFrame(() => scene?.resize())
}

const updateProjection = () => {
    if (!scene) return
    const camera = scene.camera
    const near = camera.projection.near
    const far = camera.projection.far
    scene.setCamera({
        ...camera,
        projection: projection.value === 'perspective'
            ? { type: 'perspective', fovYDegrees: 45, near, far }
            : { type: 'orthographic', height: 1.4, near, far },
    })
}

const updateSliceTool = () => scene?.setSliceTool(sliceTool.value === 'none' ? null : sliceTool.value)

const updateTime = async () => {
    scene?.setTimeIndex(timeIndex.value)
    if (profileText.value !== '尚未生成') await generateRouteProfile()
}

const toggleSlice = (axis: VolumeSliceAxis) => {
    if (!scene) return
    const existingId = lockedSliceIds[axis]
    if (existingId) {
        scene.removeSlice(existingId)
        lockedSliceIds[axis] = null
        return
    }
    lockedSliceIds[axis] = scene.lockSlice(axis, axisPositions[axis]).id
}

const moveSlice = (axis: VolumeSliceAxis) => {
    const id = lockedSliceIds[axis]
    if (id) scene?.updateSlice(id, axisPositions[axis])
}

const beginSliceInteraction = () => scene?.beginInteraction('slice')
const endSliceInteraction = () => scene?.endInteraction('slice')

const getAxisMinimum = (axis: VolumeSliceAxis) => {
    return axis === 'x' ? clip.minX : axis === 'y' ? clip.minY : clip.minZ
}

const getAxisMaximum = (axis: VolumeSliceAxis) => {
    return axis === 'x' ? clip.maxX : axis === 'y' ? clip.maxY : clip.maxZ
}

const clearSlices = () => {
    scene?.clearLockedSlices()
    lockedSliceIds.x = null
    lockedSliceIds.y = null
    lockedSliceIds.z = null
}

const applyClip = () => {
    scene?.update({ clipBox: { ...clip } })
}

const generateRouteProfile = async () => {
    if (!scene || !source) return
    const revision = ++profileRevision
    status.value = '正在生成路线剖面'
    const section = await createVolumeSection(source, {
        timeIndex: timeIndex.value,
        route: [{ ...routeStart }, { ...routeEnd }],
        sampleCount: Math.max(2, Math.round(routeSampleCount.value)),
        sampleMode: routeSampleMode.value,
    })
    if (revision !== profileRevision || !scene) return
    const profile = createVolumeSectionRouteProfile(section, {
        id: 'standalone-route-profile',
        opacity: 1,
        depthMode: 'scene',
        revision,
    })
    currentProfile = profile
    scene.setRouteProfiles([
        profile,
    ])
    profileText.value = JSON.stringify({
        id: profile.id,
        width: section.width,
        height: section.height,
        totalDistanceKm: Number(section.totalDistanceKm.toFixed(2)),
        levels: [section.heights[0], section.heights[section.heights.length - 1]],
        valueCount: section.values.length,
        valuesShared: profile.values === section.values,
    }, null, 4)
    status.value = `路线剖面已生成：${section.width}×${section.height}`
}

const handleSlicesChange = (slices: readonly IVolumeLockedSlice[]) => {
    slicesText.value = JSON.stringify(slices, null, 4)
    for (const axis of ['x', 'y', 'z'] as const) {
        const id = lockedSliceIds[axis]
        if (!id) continue
        const current = slices.find(item => item.id === id)
        if (current) axisPositions[axis] = current.position
        else lockedSliceIds[axis] = null
    }
}

const createScene = async (camera?: IVolumeCameraState) => {
    if (!canvas.value || !overlay.value || !source) return
    const activeAxes = (['x', 'y', 'z'] as const).filter(axis => lockedSliceIds[axis])
    const physicalLayout = createPhysicalVolumeLayout(source)
    scene?.destroy()
    lockedSliceIds.x = null
    lockedSliceIds.y = null
    lockedSliceIds.z = null
    scene = await createVolumeScene({
        canvas: canvas.value,
        overlayContainer: overlay.value,
        interactionElement: canvas.value,
        source,
        colorStops: volumeColorStops,
        opacityStops: volumeOpacityStops,
        colorMode: colorMode.value,
        timeIndex: timeIndex.value,
        mode: mode.value,
        quality: quality.value,
        renderScale: 1,
        opacity: opacity.value,
        densityScale: density.value,
        verticalScale: verticalScale.value + 1,
        interactionRenderScale: 0.6,
        modelMatrix: physicalLayout.modelMatrix,
        camera: camera ?? createPhysicalVolumeCamera(physicalLayout),
        chunk: {
            memoryLimitBytes: Math.max(128, Math.round(memoryLimitMb.value)) * 1024 * 1024,
        },
        sliceInteraction: {
            maxLockedSlices: Math.max(1, Math.round(maxLockedSlices.value)),
            volumeDimFactor: 0.3,
            depthMode: 'always-visible',
            plateColor: [1, 1, 1, 0.15],
            onSlicesChange: handleSlicesChange,
            onSliceError: error => {
                status.value = error.message
            },
        },
    })
    updateSliceTool()
    if (currentProfile) scene.setRouteProfiles([currentProfile])
    for (const axis of activeAxes) toggleSlice(axis)
}

const recreateScene = async () => {
    if (!scene) return
    const camera = scene.camera
    status.value = '正在重建体积资源'
    await createScene(camera)
    status.value = '体积资源配置已重建'
}

const loadArchive = async (useDefault = false) => {
    try {
        status.value = '正在加载真实 ZIP'
        const file = fileInput.value?.files?.[0]
        const loaded = useDefault || !file ? await loadDemoVolume() : await loadVolumeFile(file)
        source = loaded.source
        times.value = loaded.metadata.timeList
        timeIndex.value = 0
        currentProfile = null
        profileText.value = '尚未生成'
        const { xAxis, yAxis, zAxis } = loaded.source.metadata
        Object.assign(clip, {
            minX: Math.min(xAxis.start, xAxis.end),
            maxX: Math.max(xAxis.start, xAxis.end),
            minY: Math.min(yAxis.start, yAxis.end),
            maxY: Math.max(yAxis.start, yAxis.end),
            minZ: Math.min(zAxis.start, zAxis.end),
            maxZ: Math.max(zAxis.start, zAxis.end),
        })
        axisPositions.x = (xAxis.start + xAxis.end) / 2
        axisPositions.y = (yAxis.start + yAxis.end) / 2
        axisPositions.z = (zAxis.start + zAxis.end) / 2
        axisSteps.x = xAxis.count > 1 ? Math.abs(xAxis.end - xAxis.start) / (xAxis.count - 1) : 1
        axisSteps.y = 1
        axisSteps.z = zAxis.count > 1 ? Math.abs(zAxis.end - zAxis.start) / (zAxis.count - 1) : 1
        metadataText.value = JSON.stringify({
            sourceName: loaded.metadata.sourceName,
            productTime: loaded.metadata.productTime,
            axes: loaded.source.metadata,
            layout: 'index = ((z * height) + y) * width + x',
        }, null, 4)
        await createScene()
        status.value = `真实体数据 ${loaded.source.metadata.xAxis.count}×${loaded.source.metadata.zAxis.count}×${loaded.source.metadata.yAxis.count}`
    } catch (error) {
        status.value = error instanceof Error ? error.message : String(error)
    }
}

const start = async () => {
    if (!canvas.value || !overlay.value) return
    try {
        await loadArchive(true)
        resizeObserver = new ResizeObserver(() => scene?.resize())
        resizeObserver.observe(canvas.value)
        timer = window.setInterval(() => {
            if (!scene) return
            const statistics = scene.statistics
            const frameState = statistics.estimatedFps > 0
                ? `${statistics.frameTimeMilliseconds.toFixed(1)} ms · ${statistics.estimatedFps.toFixed(0)} FPS`
                : '空闲（按需渲染）'
            performanceText.value = `${frameState} · ${statistics.residentBrickCount}/${statistics.logicalBrickCount} 分块 · ${(statistics.gpuBytes / 1024 / 1024).toFixed(1)} MiB`
            statisticsText.value = JSON.stringify(statistics, null, 4)
        }, 500)
    } catch (error) {
        status.value = error instanceof Error ? error.message : String(error)
    }
}

onMounted(start)
onBeforeUnmount(() => {
    window.clearInterval(timer)
    resizeObserver?.disconnect()
    scene?.destroy()
    scene = null
    source = null
})
</script>

创建参数 ​

参数类型默认值必填说明
canvasHTMLCanvasElement-是WebGPU 输出画布
sourceIVolumeSource-是分块体数据源
colorStopsVolumeColorStopInput[]-是颜色传递函数
opacityStopsIVolumeOpacityStop[]-是透明度传递函数
colorMode'step' | 'smooth''step'否分段或连续颜色
mode'composite' | 'maximum''composite'否合成或最大值投影
quality'auto' | 'performance' | 'balanced' | 'quality''auto'否质量预设
opacitynumber1否整体透明度
densityScalenumber1否体密度倍率
verticalScalenumber1否垂直比例
interactionRenderScalenumber内部值否交互中的动态分辨率
chunkIVolumeChunkOptions内部值否分块尺寸和显存预算
sliceInteractionIVolumeSliceInteractionOptions-否锁定切片配置

场景方法 ​

方法说明
update(options)实时更新质量、透明度、密度、色标、裁剪盒等
setTimeIndex(index) / setSource(source)切换时间或数据源
setRouteProfiles(profiles)设置路线剖面
lockSlice(axis, position)锁定 X、Y 或 Z 切片
updateSlice(id, position) / removeSlice(id)移动或删除切片
beginInteraction(kind) / endInteraction(kind)标记交互区间，启用动态质量策略
resize() / render()调整尺寸或主动请求渲染
destroy()释放 GPU 与事件资源

scene.statistics 可读取常驻分块、上传批次、显存、帧耗时、估算 FPS、实际质量与交互状态。
