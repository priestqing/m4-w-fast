Mars3D 三维剖面墙 ​

本节由codex施工

三维剖面墙仍由 createVolumeSection 负责采样。显示端把 values 转为纹理，按路线和高度范围放到 Mars3D 墙体上；透明度和颜色只更新材质，不重新解析数据。

运行示例 ​

示例默认加载真实 ZIP，并直接生成水平层与三维剖面墙。

新窗口打开完整示例

关键代码 ​

<script setup lang="ts">
// 将公共剖面结果转换为纹理并挂到 Mars3D 的三维墙体上。
import { onBeforeUnmount, onMounted, ref } from 'vue'
import {
    calculateGridLayerStatistics,
    createHorizontalRasterGrid,
    createVolumeSection,
    createVolumeSectionRasterGrid,
} from 'm4-w-fast/volume-section'
import type { IVolumeSection } from 'm4-w-fast/volume-section'
import {
    normalizeColorStops,
    WebgpuRasterRenderer,
} from 'm4-w-fast/raster-overlay'
import { createMars3dRasterPlaneLayer } from 'm4-w-fast/raster-mars3d'
import type {
    IMars3dRasterPlaneLayer,
    IMars3dRuntime,
} from 'm4-w-fast/raster-mars3d'
import type { IVolumeSource } from 'm4-w-fast/volume-scene'
import {
    createMars3dDocumentMap,
    frameMars3dVolume,
    loadMars3dDemoRuntime,
} from './mars3dDemoRuntime'
import type { IMars3dDocumentMap, IMars3dMapClickEvent } from './mars3dDemoRuntime'
import {
    defaultSectionRoute,
    legendBackground,
    loadDemoVolume,
    loadVolumeFile,
    volumeColorStops,
} from './verticalSectionDemoData'

const container = ref<HTMLDivElement>()
const sectionCanvas = ref<HTMLCanvasElement>()
const fileInput = ref<HTMLInputElement>()
const timeIndex = ref(0)
const times = ref<readonly string[]>([])
const sampleCount = ref(400)
const sampleMode = ref<'cell' | 'interpolate'>('interpolate')
const colorMode = ref<'step' | 'smooth'>('step')
const levelIndex = ref(0)
const levels = ref<number[]>([])
const horizontalText = ref('尚无水平层结果')
const sectionText = ref('尚无剖面结果')
const metadataText = ref('尚无解析结果')
const routeStart = ref<{ longitude: number, latitude: number }>({ ...defaultSectionRoute[0] })
const routeEnd = ref<{ longitude: number, latitude: number }>({ ...defaultSectionRoute[1] })
const selectionEnabled = ref(false)
const status = ref('正在生成真实剖面')
interface MarsGraphicLayer {
    addGraphic: (graphic: unknown) => void
    clear: () => void
}
let map: IMars3dDocumentMap | null = null
let wallLayer: MarsGraphicLayer | null = null
let annotationLayer: MarsGraphicLayer | null = null
let wall: { remove: () => void } | null = null
let wallTextureUrl: string | null = null
let sectionRenderer: WebgpuRasterRenderer | null = null
let sectionRendererInitialization: Promise<void> | null = null
let sectionTextureWidth = 0
let sectionTextureHeight = 0
let wallResampleRevision = 0
let wallResampleQueue = Promise.resolve()
let horizontalLayer: IMars3dRasterPlaneLayer | null = null
let section: IVolumeSection | null = null
let source: IVolumeSource | null = null
let runtime: Awaited<ReturnType<typeof loadMars3dDemoRuntime>> | null = null
let revision = 0
let selectionStep = 0

const getRoute = () => [{ ...routeStart.value }, { ...routeEnd.value }]

const canvasToBlobUrl = (canvas: HTMLCanvasElement) => new Promise<string>((resolve, reject) => {
    canvas.toBlob((blob) => {
        if (blob) resolve(URL.createObjectURL(blob))
        else reject(new Error('无法导出剖面墙纹理'))
    }, 'image/png')
})

const calculateSectionTextureSize = (currentSection: IVolumeSection) => {
    if (!map || !runtime) return { width: 1600, height: 512 }
    const sceneCanvas = map.scene.canvas
    const bottomHeight = Math.min(...currentSection.heights)
    const topHeight = Math.max(...currentSection.heights)
    const sampleStep = Math.max(1, Math.ceil(currentSection.width / 64))
    const sampleIndexes = [
        ...new Set([
            ...Array.from(
                { length: Math.ceil(currentSection.width / sampleStep) },
                (_, index) => index * sampleStep,
            ),
            currentSection.width - 1,
        ]),
    ]
    const screenPoints = sampleIndexes.flatMap((index) => {
        return [bottomHeight, topHeight].map((height) => {
            const position = runtime?.Cesium.Cartesian3.fromDegrees(
                currentSection.longitudes[index],
                currentSection.latitudes[index],
                height,
            )
            return position
                ? runtime?.Cesium.SceneTransforms.worldToWindowCoordinates(map?.scene, position)
                : undefined
        })
    }).filter((point): point is { x: number, y: number } => Boolean(point))
    let cssWidth = sceneCanvas.clientWidth
    let cssHeight = sceneCanvas.clientHeight
    if (screenPoints.length >= 2) {
        const xValues = screenPoints.map(point => point.x)
        const yValues = screenPoints.map(point => point.y)
        cssWidth = Math.max(
            1,
            Math.min(sceneCanvas.clientWidth, Math.max(...xValues))
            - Math.max(0, Math.min(...xValues)),
        )
        cssHeight = Math.max(
            1,
            Math.min(sceneCanvas.clientHeight, Math.max(...yValues))
            - Math.max(0, Math.min(...yValues)),
        )
    }
    const ratio = (window.devicePixelRatio || 1) * 1.5
    let width = Math.max(currentSection.width * 4, Math.ceil(cssWidth * ratio))
    let height = Math.max(currentSection.height * 16, Math.ceil(cssHeight * ratio))
    const scale = Math.min(
        1,
        4096 / width,
        2048 / height,
        Math.sqrt(12000000 / (width * height)),
    )
    width = Math.max(currentSection.width, Math.round(width * scale))
    height = Math.max(currentSection.height, Math.round(height * scale))
    return { width, height }
}

const initializeSectionRenderer = async () => {
    if (sectionRenderer?.state === 'ready') return
    if (sectionRendererInitialization) return sectionRendererInitialization
    if (!sectionCanvas.value) throw new Error('剖面墙画布尚未挂载')
    const renderer = new WebgpuRasterRenderer()
    sectionRenderer = renderer
    sectionRendererInitialization = renderer.initialize(sectionCanvas.value, 'simple').then(() => {
        renderer.setColorStops(normalizeColorStops([...volumeColorStops]))
        renderer.setColorRange({ minValue: 10, maxValue: 80 })
        renderer.setOpacity(1)
        renderer.setSampleMode('interpolate')
    })
    try {
        await sectionRendererInitialization
    } finally {
        sectionRendererInitialization = null
    }
}

const renderWall = async (force = true) => {
    if (!runtime || !section || !wallLayer || !sectionCanvas.value) return
    const renderRevision = revision
    await initializeSectionRenderer()
    if (!sectionRenderer || renderRevision !== revision) return
    const grid = createVolumeSectionRasterGrid(section)
    const { width: textureWidth, height: textureHeight } = calculateSectionTextureSize(section)
    if (
        !force
        && sectionTextureWidth > 0
        && sectionTextureHeight > 0
        && Math.abs(textureWidth - sectionTextureWidth) / sectionTextureWidth < 0.2
        && Math.abs(textureHeight - sectionTextureHeight) / sectionTextureHeight < 0.2
    ) return
    sectionCanvas.value.width = textureWidth
    sectionCanvas.value.height = textureHeight
    sectionRenderer.setGrid({
        ...grid,
        minLng: 0,
        maxLng: textureWidth,
        minLat: 0,
        maxLat: textureHeight,
    })
    sectionRenderer.setColorMode(colorMode.value)
    await Promise.resolve(sectionRenderer.render({
        canvasWidth: textureWidth,
        canvasHeight: textureHeight,
        cssWidth: textureWidth,
        cssHeight: textureHeight,
        devicePixelRatio: 1,
        zoom: 0,
        worldTopLeftX: 0,
        worldTopLeftY: -textureHeight,
    }))
    await new Promise<void>(resolve => requestAnimationFrame(() => requestAnimationFrame(() => resolve())))
    const textureUrl = await canvasToBlobUrl(sectionCanvas.value)
    if (renderRevision !== revision) {
        URL.revokeObjectURL(textureUrl)
        return
    }
    const bottomHeight = Math.min(...section.heights)
    const topHeight = Math.max(...section.heights)
    const positions = Array.from({ length: section.width }, (_, index) => [
        section?.longitudes[index],
        section?.latitudes[index],
        bottomHeight,
    ])
    const nextWall = new runtime.mars3d.graphic.WallPrimitive({
        positions,
        style: {
            diffHeight: topHeight - bottomHeight,
            image: textureUrl,
            opacity: 0.96,
            flat: true,
            outline: true,
            outlineColor: '#d9e8f7',
            outlineOpacity: 0.55,
        },
    })
    wallLayer.addGraphic(nextWall)
    wall?.remove()
    wall = nextWall
    if (wallTextureUrl) URL.revokeObjectURL(wallTextureUrl)
    wallTextureUrl = textureUrl
    sectionTextureWidth = textureWidth
    sectionTextureHeight = textureHeight
    annotationLayer?.clear()
    annotationLayer?.addGraphic(new runtime.mars3d.graphic.PolylinePrimitive({
        positions: Array.from({ length: section.width }, (_, index) => [
            section?.longitudes[index],
            section?.latitudes[index],
            0,
        ]),
        style: { width: 4, color: '#ffffff', opacity: 0.82, clampToGround: true },
    }))
    const start = section.routeVertices[0]
    const end = section.routeVertices[section.routeVertices.length - 1]
    for (const [point, text, color] of [
        [start, 'A 起点', '#39e58c'],
        [end, 'B 终点', '#ff626d'],
    ] as const) {
        annotationLayer?.addGraphic(new runtime.mars3d.graphic.PointPrimitive({
            position: [point.longitude, point.latitude, 0],
            style: {
                pixelSize: 10,
                color,
                outline: true,
                outlineColor: '#ffffff',
                outlineWidth: 2,
                label: { text, color: '#ffffff', pixelOffsetY: -24 },
            },
        }))
    }
    sectionText.value = JSON.stringify({
        timeIndex: section.timeIndex,
        sampleMode: section.sampleMode,
        route: {
            start,
            end,
            totalDistanceKm: Number(section.totalDistanceKm.toFixed(3)),
        },
        matrix: {
            width: section.width,
            height: section.height,
            textureWidth,
            textureHeight,
        },
        renderer: 'Mars3D WallPrimitive + WebgpuRasterRenderer',
        ...calculateGridLayerStatistics(section.values),
    }, null, 4)
}

const requestWallRender = (force = false) => {
    const requestRevision = ++wallResampleRevision
    wallResampleQueue = wallResampleQueue.catch(() => undefined).then(async () => {
        if (requestRevision !== wallResampleRevision) return
        await renderWall(force)
    })
    return wallResampleQueue
}

const generate = async () => {
    if (!runtime || !source) return
    const currentRevision = ++revision
    const startedAt = performance.now()
    const result = await createVolumeSection(source, {
        timeIndex: timeIndex.value,
        route: getRoute(),
        sampleCount: sampleCount.value,
        sampleMode: sampleMode.value,
    })
    if (currentRevision !== revision) return
    section = result
    await requestWallRender(true)
    fitSectionView()
    status.value = `${result.width}×${result.height} · ${result.totalDistanceKm.toFixed(1)} km · ${(performance.now() - startedAt).toFixed(1)} ms`
}

const renderHorizontalLayer = async () => {
    if (!map || !runtime || !source) return
    const grid = await createHorizontalRasterGrid(source, timeIndex.value, levelIndex.value)
    const options = {
        grid,
        height: levels.value[levelIndex.value],
        colorStops: [...volumeColorStops],
        colorRange: {
            minValue: 10,
            maxValue: 80,
        },
        colorMode: colorMode.value,
        sampleMode: sampleMode.value,
        opacity: 0.5,
    }
    if (horizontalLayer) horizontalLayer.update(options)
    else {
        horizontalLayer = createMars3dRasterPlaneLayer(
            runtime.mars3d as unknown as IMars3dRuntime,
            { name: '水平高度栅格', ...options },
        )
        map.addLayer(horizontalLayer)
    }
    horizontalText.value = JSON.stringify({
        timeIndex: timeIndex.value,
        levelIndex: levelIndex.value,
        levelValue: levels.value[levelIndex.value],
        grid: { width: grid.width, height: grid.height },
        ...calculateGridLayerStatistics(grid.values),
    }, null, 4)
}

const renderComparison = async () => {
    await generate()
    await renderHorizontalLayer()
}

const loadArchive = async () => {
    try {
        status.value = '正在加载真实 ZIP'
        const file = fileInput.value?.files?.[0]
        const loaded = file ? await loadVolumeFile(file) : await loadDemoVolume()
        source = loaded.source
        times.value = loaded.metadata.timeList
        levels.value = [...loaded.metadata.levelList]
        timeIndex.value = Math.min(timeIndex.value, times.value.length - 1)
        levelIndex.value = Math.min(levelIndex.value, levels.value.length - 1)
        metadataText.value = JSON.stringify({
            sourceName: loaded.metadata.sourceName,
            productTime: loaded.metadata.productTime,
            axes: loaded.source.metadata,
            layout: 'index = ((z * height) + y) * width + x',
        }, null, 4)
        resetView()
        await renderComparison()
    } catch (error) {
        status.value = error instanceof Error ? error.message : String(error)
    }
}

const findSectionPeakSampleIndex = (currentSection: IVolumeSection) => {
    let peakValue = Number.NEGATIVE_INFINITY
    let peakSampleIndex = Math.floor(currentSection.width / 2)
    currentSection.values.forEach((value, index) => {
        if (!Number.isFinite(value) || value <= peakValue) return
        peakValue = value
        peakSampleIndex = index % currentSection.width
    })
    return peakSampleIndex
}

const calculateSectionFacingHeading = (
    currentSection: IVolumeSection,
    startIndex: number,
    endIndex: number,
) => {
    const toRadians = (value: number) => value * Math.PI / 180
    const startLongitude = toRadians(currentSection.longitudes[startIndex])
    const startLatitude = toRadians(currentSection.latitudes[startIndex])
    const endLongitude = toRadians(currentSection.longitudes[endIndex])
    const endLatitude = toRadians(currentSection.latitudes[endIndex])
    const deltaLongitude = endLongitude - startLongitude
    const y = Math.sin(deltaLongitude) * Math.cos(endLatitude)
    const x = Math.cos(startLatitude) * Math.sin(endLatitude)
        - Math.sin(startLatitude) * Math.cos(endLatitude) * Math.cos(deltaLongitude)
    return Math.atan2(y, x) + Math.PI / 2
}

const fitSectionView = () => {
    if (!map || !runtime || !section || !wall) return
    const peakSampleIndex = findSectionPeakSampleIndex(section)
    const halfWindow = Math.max(8, Math.round(section.width * 0.02))
    const startIndex = Math.max(0, peakSampleIndex - halfWindow)
    const endIndex = Math.min(section.width - 1, peakSampleIndex + halfWindow)
    const bottomHeight = Math.min(...section.heights)
    const topHeight = Math.max(...section.heights)
    const sampleStep = Math.max(1, Math.ceil((endIndex - startIndex + 1) / 16))
    const focusIndexes = [
        ...new Set([
            ...Array.from(
                { length: Math.ceil((endIndex - startIndex + 1) / sampleStep) },
                (_, index) => startIndex + index * sampleStep,
            ),
            endIndex,
        ]),
    ]
    const points = focusIndexes.flatMap(index => [bottomHeight, topHeight].map((height) => {
        return runtime?.Cesium.Cartesian3.fromDegrees(
            section?.longitudes[index] ?? 0,
            section?.latitudes[index] ?? 0,
            height,
        )
    }))
    const sphere = runtime.Cesium.BoundingSphere.fromPoints(points)
    const heading = calculateSectionFacingHeading(section, startIndex, endIndex)
    const offset = new runtime.Cesium.HeadingPitchRange(
        heading,
        0,
        Math.max(25000, sphere.radius * 2.2),
    )
    map.scene.camera.cancelFlight()
    map.scene.camera.viewBoundingSphere(sphere, offset)
    map.scene.camera.lookAtTransform(runtime.Cesium.Matrix4.IDENTITY)
}

const resetView = () => {
    if (section) fitSectionView()
    else if (map && runtime && source) frameMars3dVolume(map, runtime, source, 1)
}

const handleCameraMoveEnd = () => {
    void requestWallRender(false).catch((error) => {
        status.value = error instanceof Error ? error.message : String(error)
    })
}

const toggleSelection = () => {
    selectionEnabled.value = !selectionEnabled.value
    selectionStep = 0
    status.value = selectionEnabled.value
        ? '请在地图依次选择 A、B'
        : '地图选点已关闭'
}

const handleMapSelection = (event: IMars3dMapClickEvent) => {
    if (!selectionEnabled.value || !runtime || !source || !event.cartesian) return
    const selected = runtime.mars3d.LngLatPoint.fromCartesian(event.cartesian)
    const { xAxis, zAxis } = source.metadata
    const minLongitude = Math.min(xAxis.start, xAxis.end)
    const maxLongitude = Math.max(xAxis.start, xAxis.end)
    const minLatitude = Math.min(zAxis.start, zAxis.end)
    const maxLatitude = Math.max(zAxis.start, zAxis.end)
    if (
        selected.lng < minLongitude
        || selected.lng > maxLongitude
        || selected.lat < minLatitude
        || selected.lat > maxLatitude
    ) {
        status.value = '请选择体数据范围内的位置'
        return
    }
    const point = { longitude: selected.lng, latitude: selected.lat }
    if (selectionStep === 0) {
        routeStart.value = point
        selectionStep = 1
        annotationLayer?.clear()
        annotationLayer?.addGraphic(new runtime.mars3d.graphic.PointPrimitive({
            position: [point.longitude, point.latitude, 0],
            style: {
                pixelSize: 10,
                color: '#39e58c',
                outline: true,
                outlineColor: '#ffffff',
                outlineWidth: 2,
                label: { text: 'A 起点', color: '#ffffff', pixelOffsetY: -24 },
            },
        }))
        status.value = `起点 A 已选择：${point.longitude.toFixed(4)}, ${point.latitude.toFixed(4)}；请继续选择 B`
        return
    }
    routeEnd.value = point
    selectionStep = 0
    selectionEnabled.value = false
    void generate()
}

const start = async () => {
    if (!container.value) return
    try {
        runtime = await loadMars3dDemoRuntime()
        map = createMars3dDocumentMap(container.value, runtime)
        wallLayer = new runtime.mars3d.layer.GraphicLayer({ name: '剖面墙' })
        annotationLayer = new runtime.mars3d.layer.GraphicLayer({ name: '剖面标注' })
        map.addLayer(wallLayer)
        map.addLayer(annotationLayer)
        map.on(runtime.mars3d.EventType.click, handleMapSelection)
        map.on(runtime.mars3d.EventType.cameraMoveEnd, handleCameraMoveEnd)
        await loadArchive()
    } catch (error) {
        status.value = error instanceof Error ? error.message : String(error)
    }
}

onMounted(start)
onBeforeUnmount(() => {
    revision += 1
    if (map && runtime) map.off(runtime.mars3d.EventType.click, handleMapSelection)
    if (map && runtime) map.off(runtime.mars3d.EventType.cameraMoveEnd, handleCameraMoveEnd)
    sectionRenderer?.destroy()
    sectionRenderer = null
    wall?.remove()
    wall = null
    if (wallTextureUrl) URL.revokeObjectURL(wallTextureUrl)
    wallTextureUrl = null
    map?.destroy()
    horizontalLayer = null
    wallLayer = null
    annotationLayer = null
    source = null
    map = null
})
</script>

                <h2>当前水平层</h2><pre id="horizontalOutput">{{ horizontalText }}</pre>

剖面结果字段 ​

字段类型说明
valuesFloat32ArrayX 最快排列，索引为 level * width + sample
heightsFloat32Array每一行对应的真实高度
longitudes / latitudesFloat64Array每个采样列的位置
distancesKmFloat64Array从路线起点开始的累计距离
routeVerticesIVolumeSectionRouteVertex[]输入折点到采样列的映射
noDatanumber剖面内部缺测标记；显示前仍应检查非有限值

使用要点 ​

  • 路线折点数量不限，但必须至少包含两个有效经纬度点。
  • sampleCount 控制纹理宽度；高度层数量决定纹理高度。
  • 拖动采样数量会重新读取并生成剖面，透明度和色标模式只重绘现有结果。
  • Mars3D/Cesium 负责墙体几何和相机，体数据解析与路线采样不依赖宿主。
