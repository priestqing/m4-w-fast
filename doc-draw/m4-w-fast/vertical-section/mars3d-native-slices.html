Mars3D 原生三维体积（不流畅待优化） ​

本节由codex施工

Mars3D 三维方案使用 Cesium VoxelPrimitive 显示体数据，并用 Cesium 几何显示 X、Y、Z 轴切片。体积、切片、深度、遮挡和生命周期都由 Cesium 原生渲染链管理。

运行示例 ​

示例默认加载真实 ZIP，可直接显示、移动或删除 X、Y、Z 轴切片。

新窗口打开完整示例

关键代码 ​

vue
<script setup lang="ts">
// 使用 Cesium 原生体素与原生几何显示真实体数据及轴向切片。
import { onBeforeUnmount, onMounted, reactive, ref } from 'vue'
import { createMars3dNativeSliceVolumeLayer } from 'm4-w-fast/volume-mars3d-native'
import type { IMars3dNativeSliceVolumeLayer } from 'm4-w-fast/volume-mars3d-native'
import type { VolumeQualityPreset, VolumeSliceAxis } from 'm4-w-fast/volume-scene'
import {
    createMars3dDocumentMap,
    frameMars3dVolume,
    loadMars3dDemoRuntime,
} from './mars3dDemoRuntime'
import type { IMars3dDocumentMap } from './mars3dDemoRuntime'
import { loadDemoVolume, loadVolumeFile, volumeColorStops, volumeOpacityStops } from './verticalSectionDemoData'

const container = ref<HTMLDivElement>()
const fileInput = ref<HTMLInputElement>()
const timeIndex = ref(0)
const times = ref<readonly string[]>([])
const quality = ref<VolumeQualityPreset>('balanced')
const opacity = ref(0.3)
const density = ref(100)
const verticalScale = ref(1)
const colorMode = 'step' as const
const status = ref('正在创建 Cesium 原生体积')
let map: IMars3dDocumentMap | null = null
let layer: IMars3dNativeSliceVolumeLayer | null = null
let runtime: Awaited<ReturnType<typeof loadMars3dDemoRuntime>> | null = null
let removePostRender: (() => void) | null = null
const axisPositions = reactive({ x: 101.2, y: 6000, z: 36.8 })
const axisSteps = reactive({ x: 0.01, y: 1, z: 0.01 })
const axisBounds = reactive({
    x: { min: 0, max: 1 },
    y: { min: 0, max: 1 },
    z: { min: 0, max: 1 },
})
const lockedSliceIds = reactive<Record<VolumeSliceAxis, string | null>>({
    x: null,
    y: null,
    z: null,
})

const updateLayer = () => {
    layer?.update({
        opacity: opacity.value,
        densityScale: density.value,
        verticalScale: verticalScale.value,
        colorMode,
        quality: quality.value,
    })
}

const updateTime = async () => {
    await layer?.setTimeIndex(timeIndex.value)
}

const toggleSlice = (axis: VolumeSliceAxis) => {
    if (!layer) return
    const existingId = lockedSliceIds[axis]
    if (existingId) {
        layer.removeSlice(existingId)
        lockedSliceIds[axis] = null
        return
    }
    lockedSliceIds[axis] = layer.lockSlice(axis, axisPositions[axis]).id
}

const moveSlice = (axis: VolumeSliceAxis) => {
    const id = lockedSliceIds[axis]
    if (id) layer?.updateSlice(id, axisPositions[axis])
}

const loadArchive = async (useDefault = false) => {
    if (!runtime || !map) return
    try {
        status.value = '正在加载真实 ZIP'
        const file = fileInput.value?.files?.[0]
        const loaded = useDefault || !file ? await loadDemoVolume() : await loadVolumeFile(file)
        times.value = loaded.metadata.timeList
        timeIndex.value = 0
        lockedSliceIds.x = null
        lockedSliceIds.y = null
        lockedSliceIds.z = null
        const { xAxis, yAxis, zAxis } = loaded.source.metadata
        axisPositions.x = (xAxis.start + xAxis.end) / 2
        axisPositions.y = (yAxis.start + yAxis.end) / 2
        axisPositions.z = (zAxis.start + zAxis.end) / 2
        Object.assign(axisBounds.x, { min: Math.min(xAxis.start, xAxis.end), max: Math.max(xAxis.start, xAxis.end) })
        Object.assign(axisBounds.y, { min: Math.min(yAxis.start, yAxis.end), max: Math.max(yAxis.start, yAxis.end) })
        Object.assign(axisBounds.z, { min: Math.min(zAxis.start, zAxis.end), max: Math.max(zAxis.start, zAxis.end) })
        axisSteps.x = xAxis.count > 1 ? Math.abs(xAxis.end - xAxis.start) / (xAxis.count - 1) : 1
        axisSteps.y = 1
        axisSteps.z = zAxis.count > 1 ? Math.abs(zAxis.end - zAxis.start) / (zAxis.count - 1) : 1
        layer?.destroy()
        layer = createMars3dNativeSliceVolumeLayer(runtime.mars3dNative, runtime.Cesium, {
            source: loaded.source,
            colorStops: volumeColorStops,
            opacityStops: volumeOpacityStops,
            colorMode,
            timeIndex: timeIndex.value,
            quality: quality.value,
            opacity: opacity.value,
            densityScale: density.value,
            verticalScale: verticalScale.value,
            onError: error => {
                status.value = `原生体积错误：${error.message}`
            },
        })
        map.addLayer(layer)
        await layer.readyPromise
        frameMars3dVolume(map, runtime, loaded.source, verticalScale.value, true)
        status.value = '原生体积已加载，可按需显示轴切片'
    } catch (error) {
        status.value = error instanceof Error ? error.message : String(error)
    }
}

const start = async () => {
    if (!container.value) return
    try {
        runtime = await loadMars3dDemoRuntime()
        map = createMars3dDocumentMap(container.value, runtime)
        await loadArchive(true)
        let frameCount = 0
        let frameWindowStart = performance.now()
        removePostRender = map.scene.postRender.addEventListener(() => {
            frameCount += 1
            const now = performance.now()
            if (now - frameWindowStart < 500) return
            const statistics = layer?.statistics
            if (!statistics) return
            const fps = frameCount * 1000 / (now - frameWindowStart)
            status.value = `${fps.toFixed(1)} FPS · ${(statistics.texturesByteLength / 1024 / 1024).toFixed(1)} MiB · ${statistics.numberOfTilesWithContentReady} 个就绪瓦片`
            frameCount = 0
            frameWindowStart = now
        })
    } catch (error) {
        status.value = error instanceof Error ? error.message : String(error)
    }
}

onMounted(start)
onBeforeUnmount(() => {
    removePostRender?.()
    removePostRender = null
    layer?.destroy()
    map?.destroy()
    layer = null
    map = null
    runtime = null
})
</script>

<template>
    <main>
        <div ref="container" class="mars3d-container" />
        <section class="toolbar">
            <div class="control">
                <label for="nativeFile">真实 ZIP / Real ZIP</label>
                <input id="nativeFile" ref="fileInput" type="file" accept=".zip,application/zip">
            </div>
            <div class="control">
                <label for="nativeTime">时间 / Time</label>
                <select id="nativeTime" v-model.number="timeIndex" @change="updateTime">
                    <option v-for="(time, index) in times" :key="`${time}-${index}`" :value="index">
                        {{ index }}: {{ time }}
                    </option>
                </select>
            </div>
            <div class="control">
                <label for="nativeQuality">画质 / Quality</label>
                <select id="nativeQuality" v-model="quality" @change="updateLayer">
                    <option value="balanced">均衡 / Balanced</option>
                    <option value="quality">质量 / Quality</option>
                    <option value="performance">性能 / Performance</option>
                    <option value="auto">自动 / Auto</option>
                </select>
            </div>
            <div class="control">
                <label for="nativeOpacity">透明度 / Opacity <span>{{ opacity.toFixed(2) }}</span></label>
                <input id="nativeOpacity" v-model.number="opacity" type="range" min="0" max="1" step="0.01" @input="updateLayer">
            </div>
            <div class="control">
                <label for="nativeDensity">密度 / Density <span>{{ density.toFixed(0) }}</span></label>
                <input id="nativeDensity" v-model.number="density" type="range" min="1" max="200" step="1" @input="updateLayer">
            </div>
            <div class="control">
                <label for="nativeVertical">垂直倍率 / Vertical <span>{{ verticalScale.toFixed(1) }}</span></label>
                <input id="nativeVertical" v-model.number="verticalScale" type="range" min="1" max="40" step="0.5" @input="updateLayer">
            </div>
            <div class="control">
                <button type="button" @click="loadArchive(false)">加载选择 ZIP</button>
                <button class="secondary" type="button" @click="loadArchive(true)">加载仓库真实 ZIP</button>
            </div>
            <div class="status">{{ status }}</div>
        </section>
        <section class="axis-panel">
            <h2>轴选层 / Axis slices</h2>
            <div v-for="axis in (['x', 'y', 'z'] as const)" :key="axis" class="axis-row">
                <button class="secondary" :class="{ active: lockedSliceIds[axis] }" type="button" @click="toggleSlice(axis)">
                    {{ lockedSliceIds[axis] ? `隐藏 ${axis.toUpperCase()}` : `显示 ${axis.toUpperCase()}` }}
                </button>
                <input
                    v-model.number="axisPositions[axis]"
                    type="range"
                    :min="axisBounds[axis].min"
                    :max="axisBounds[axis].max"
                    :step="axisSteps[axis]"
                    @input="moveSlice(axis)"
                >
                <output>{{ axis.toUpperCase() }} {{ axisPositions[axis].toFixed(axis === 'y' ? 0 : 4) }}{{ axis === 'y' ? ' m' : '°' }}</output>
            </div>
        </section>
        <aside class="metrics">{{ status }}<br>Slice atlas {{ lockedSliceIds.x || lockedSliceIds.y || lockedSliceIds.z ? 'active' : '--' }}</aside>
        <aside class="notice">原生切片实验：24 个实际高度面由一个 Cesium Primitive 批量绘制，与地图共用同一渲染帧，不监听相机后追赶。</aside>
    </main>
</template>

创建参数 ​

参数类型必填说明
mars3dIMars3dNativeRuntime是Mars3D 运行时
cesiumICesiumNativeRuntime是Cesium 原生构造器集合
sourceIVolumeSource是体数据源
colorStopsVolumeColorStopInput[]是颜色传递函数
opacityStopsIVolumeOpacityStop[]是透明度传递函数
qualityVolumeQualityPreset否原生体素质量
opacity / densityScale / verticalScalenumber否实时显示参数
colorMode'step' | 'smooth'否分段或连续颜色
onError(error: Error) => void否异步错误回调

图层方法 ​

方法说明
update(options)更新质量、透明度、密度、垂直比例和传递函数
setSource(source) / setTimeIndex(index)异步重建原生数据
lockSlice(axis, position)新增一个轴向切片并返回其 id
updateSlice(id, position)移动现有切片
removeSlice(id) / clearLockedSlices()删除切片
destroy()销毁体素、切片几何、材质与监听

statistics 提供纹理字节数、访问瓦片数和内容就绪瓦片数。切片位置使用源坐标：X/Z 为经纬度，Y 为高度。
