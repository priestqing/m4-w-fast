Leaflet WebGL2 ​

本节由codex施工

WebGL2 道路图层把道路 mesh、规则栅格和色标上传到 GPU，地图拖动或缩放时按当前 Leaflet 视口重绘。业务侧不需要直接管理 Canvas、Shader 或纹理。

运行示例 ​

示例实际加载 5,152 条 highway 道路和完整 MICAPS4 文件 2510270910.015。默认使用分段色标与双线性插值；透明度、线宽、颜色模式和采样模式均直接更新现有图层。色标始终显示完整色阶，被当前范围过滤的部分降低透明度。性能栏显示真实 mesh 规模、连续提交次数和下一帧响应时间。

<!-- 使用真实道路与栅格数据展示 Leaflet 道路 GPU 图层和连续参数更新。 -->
<script setup lang="ts">
import { computed, nextTick, onBeforeUnmount, onMounted, ref } from 'vue'
import * as L from 'leaflet'
import 'leaflet/dist/leaflet.css'
import {
    createRoadLeafletLayer,
    prepareRoadData,
} from 'm4-w-fast/road-overlay'
import type {
    RoadLeafletLayer,
    RoadRendererType,
} from 'm4-w-fast/road-overlay'
import type {
    RasterColorMode,
    RasterSampleMode,
} from 'm4-w-fast/raster-overlay'
import {
    loadRasterDemoGrid,
    loadRoadDemoData,
    rasterDemoColorStops,
    roadDemoBounds,
} from './roadLeafletDemoData'

const props = defineProps<{
    rendererType: RoadRendererType
}>()

const mapElement = ref<HTMLDivElement | null>(null)
const status = ref('等待初始化')
const opacity = ref(0.95)
const lineWidth = ref(5)
const colorMode = ref<RasterColorMode>('step')
const sampleMode = ref<RasterSampleMode>('interpolate')
const minValue = ref(-8)
const maxValue = ref(8)
const roadCount = ref(0)
const segmentCount = ref(0)
const triangleCount = ref(0)
const liveUpdateCount = ref(0)
const frameResponseTime = ref<number | null>(null)
const lastUpdateLabel = ref('拖动滑块后显示实时更新结果')
const backendLabel = computed(() => props.rendererType.toUpperCase())
const roadSummary = computed(() => {
    const roadsLabel = `${roadCount.value.toLocaleString()} 条道路`
    const segmentsLabel = `${segmentCount.value.toLocaleString()} 个路段`
    const trianglesLabel = `${triangleCount.value.toLocaleString()} 个三角形`
    return `${roadsLabel} · ${segmentsLabel} · ${trianglesLabel}`
})
const legendModeLabel = computed(() => colorMode.value === 'step' ? '分段着色' : '平滑着色')
const legendTicks = [-20, -12, -4, 4, 12, 20]

let map: L.Map | null = null
let roadLayer: RoadLeafletLayer | null = null
let resizeObserver: ResizeObserver | null = null
let roads: Awaited<ReturnType<typeof loadRoadDemoData>> | null = null
let grid: Awaited<ReturnType<typeof loadRasterDemoGrid>> | null = null
let rebuildFrame = 0
let rebuildVersion = 0
let performanceFrame = 0
let latestUpdateStartedAt = 0

const recordLiveUpdate = (label: string, update: () => void) => {
    latestUpdateStartedAt = performance.now()
    update()
    liveUpdateCount.value += 1
    lastUpdateLabel.value = label

    if (performanceFrame) return

    performanceFrame = requestAnimationFrame(() => {
        frameResponseTime.value = performance.now() - latestUpdateStartedAt
        performanceFrame = 0
    })
}

const readControlValue = (event: Event): string => {
    return (event.target as HTMLInputElement | HTMLSelectElement).value
}

const updateOpacity = (event: Event) => {
    const value = Number(readControlValue(event))
    opacity.value = value
    recordLiveUpdate('透明度已连续提交', () => roadLayer?.setOpacity(value))
}

const updateLineWidth = (event: Event) => {
    const value = Number(readControlValue(event))
    lineWidth.value = value
    recordLiveUpdate('线宽已连续提交', () => roadLayer?.setLineWidth(value))
}

const updateColorMode = (event: Event) => {
    const value = readControlValue(event) as RasterColorMode
    colorMode.value = value
    roadLayer?.setColorMode(value)
}

const updateSampleMode = (event: Event) => {
    const value = readControlValue(event) as RasterSampleMode
    sampleMode.value = value
    roadLayer?.setSampleMode(value)
}

const requestColorRangeUpdate = (source: 'min' | 'max', event: Event) => {
    const value = Number(readControlValue(event))
    if (source === 'min') minValue.value = value
    else maxValue.value = value

    if (minValue.value > maxValue.value) {
        if (source === 'min') maxValue.value = minValue.value
        else minValue.value = maxValue.value
    }

    recordLiveUpdate('显示值范围已连续提交', () => {
        if (rebuildFrame) cancelAnimationFrame(rebuildFrame)
        rebuildFrame = requestAnimationFrame(() => {
            rebuildFrame = 0
            void rebuildLayer()
        })
    })
}

const rebuildLayer = async () => {
    if (!map || !roads || !grid) return

    const currentMap = map
    const previousLayer = roadLayer
    const currentVersion = ++rebuildVersion

    status.value = `正在创建 ${backendLabel.value} 道路图层`
    const nextLayer = await createRoadLeafletLayer({
        rendererType: props.rendererType,
        roads,
        grid,
        colorStops: rasterDemoColorStops,
        colorRange: {
            minValue: minValue.value,
            maxValue: maxValue.value,
        },
        colorMode: colorMode.value,
        sampleMode: sampleMode.value,
        opacity: opacity.value,
        lineWidth: lineWidth.value,
        pane: 'road-demo-pane',
    })

    if (map !== currentMap || currentVersion !== rebuildVersion) {
        nextLayer.remove()
        return
    }

    nextLayer.addTo(currentMap)
    roadLayer = nextLayer
    if (previousLayer) currentMap.removeLayer(previousLayer)
    status.value = `${backendLabel.value} 已绘制 ${roadCount.value.toLocaleString()} 条道路`
}

const initialize = async () => {
    if (!mapElement.value) return

    try {
        if (props.rendererType === 'webgpu') {
            if (!navigator.gpu) throw new Error('当前浏览器不支持 WebGPU')
            const adapter = await navigator.gpu.requestAdapter()
            if (!adapter) throw new Error('浏览器未能取得 WebGPU Adapter')
        }

        status.value = '正在加载真实道路和栅格数据'
        const loadedData = await Promise.all([
            loadRoadDemoData(),
            loadRasterDemoGrid(),
        ])
        roads = loadedData[0]
        grid = loadedData[1]
        const prepared = prepareRoadData(roads)
        roadCount.value = prepared.geometry.lineCount
        segmentCount.value = prepared.mesh.segments.length
        triangleCount.value = prepared.mesh.triangleCount

        map = L.map(mapElement.value, {
            center: [35.97, 103.75],
            zoom: 10,
            zoomSnap: 0.25,
        })
        L.tileLayer('https://webrd0{s}.is.autonavi.com/appmaptile?lang=zh_cn&size=1&scale=1&style=8&x={x}&y={y}&z={z}', {
            subdomains: ['1', '2', '3', '4'],
            maxZoom: 18,
        }).addTo(map)
        map.createPane('road-demo-pane')
        const pane = map.getPane('road-demo-pane')
        if (pane) pane.style.zIndex = '420'

        resizeObserver = new ResizeObserver(() => map?.invalidateSize())
        resizeObserver.observe(mapElement.value)
        await rebuildLayer()
        map.fitBounds(roadDemoBounds, { padding: [20, 20] })
        await nextTick()
        map.invalidateSize()
    } catch (error) {
        status.value = error instanceof Error ? error.message : '示例初始化失败'
    }
}

onMounted(() => void initialize())

onBeforeUnmount(() => {
    rebuildVersion += 1
    if (rebuildFrame) cancelAnimationFrame(rebuildFrame)
    rebuildFrame = 0
    if (performanceFrame) cancelAnimationFrame(performanceFrame)
    performanceFrame = 0
    resizeObserver?.disconnect()
    resizeObserver = null
    map?.remove()
    roadLayer = null
    map = null
})
</script>

<template>
    <section class="road-demo">
        <header class="demo-header">
            <div>
                <h3>Leaflet {{ backendLabel }} 道路</h3>
                <p>{{ status }}</p>
            </div>
            <span class="backend-badge">{{ props.rendererType }}</span>
        </header>

        <div class="controls">
            <label>
                <span>颜色模式</span>
                <select :value="colorMode" @change="updateColorMode">
                    <option value="step">分段</option>
                    <option value="smooth">平滑</option>
                </select>
            </label>
            <label>
                <span>采样模式</span>
                <select :value="sampleMode" @change="updateSampleMode">
                    <option value="interpolate">双线性插值</option>
                    <option value="cell">原始格点</option>
                </select>
            </label>
            <label>
                <span>透明度 {{ opacity.toFixed(2) }}</span>
                <input :value="opacity" type="range" min="0" max="1" step="0.01" @input="updateOpacity">
            </label>
            <label>
                <span>线宽 {{ lineWidth.toFixed(1) }}px</span>
                <input :value="lineWidth" type="range" min="1" max="16" step="0.5" @input="updateLineWidth">
            </label>
            <label>
                <span>最小值 {{ minValue }}</span>
                <input :value="minValue" type="range" min="-20" max="12" step="1" @input="requestColorRangeUpdate('min', $event)">
            </label>
            <label>
                <span>最大值 {{ maxValue }}</span>
                <input :value="maxValue" type="range" min="-8" max="20" step="1" @input="requestColorRangeUpdate('max', $event)">
            </label>
        </div>

        <div class="legend" aria-label="道路色标">
            <div class="legend-heading">
                <strong>色标</strong>
                <span>{{ legendModeLabel }} · {{ rasterDemoColorStops.length }} 个色阶</span>
            </div>
            <div class="legend-colors">
                <span
                    v-for="(stop, index) in rasterDemoColorStops"
                    :key="`${stop.min}-${stop.max}-${index}`"
                    :style="{ backgroundColor: stop.color }"
                    :class="{ 'is-filtered': stop.max <= minValue || stop.min >= maxValue }"
                    :title="`${stop.min} ～ ${stop.max}`"
                />
            </div>
            <div class="legend-ticks">
                <span v-for="tick in legendTicks" :key="tick">{{ tick }}</span>
            </div>
        </div>

        <div class="performance-panel">
            <div>
                <strong>GPU 实时重绘</strong>
                <span>{{ roadSummary }}</span>
            </div>
            <div>
                <strong>{{ liveUpdateCount }} 次</strong>
                <span>{{ lastUpdateLabel }}</span>
            </div>
            <div>
                <strong>{{ frameResponseTime === null ? '--' : `${frameResponseTime.toFixed(1)} ms` }}</strong>
                <span>下一帧响应</span>
            </div>
            <p>快速拖动时，连续参数更新会合并到浏览器下一渲染帧，避免同一帧重复绘制。</p>
        </div>

        <div ref="mapElement" class="map" />
    </section>
</template>

// 提供道路文档使用的真实 highway GeoJSON、行政区边界和栅格数据。
import type { FeatureCollection, MultiPolygon, Polygon } from 'geojson'
import type { RoadGeoJsonInput } from 'm4-w-fast/road-overlay'
import roadGeoJsonUrl from './data/road-highways.json?url'
import boundaryGeoJsonUrl from '../raster/data/620000.json?url'
import {
    loadRasterDemoGrid,
    rasterDemoColorStops,
} from '../raster/rasterLeafletDemoData'

export const roadDemoBounds: [[number, number], [number, number]] = [
    [35.8304181, 103.6126398],
    [36.1099785, 103.9042068],
]

const loadJson = async <T>(url: string, label: string): Promise<T> => {
    const response = await fetch(url)
    if (!response.ok) throw new Error(`${label} 加载失败：${response.status}`)
    return await response.json() as T
}

export const loadRoadDemoData = (): Promise<RoadGeoJsonInput> => {
    return loadJson<RoadGeoJsonInput>(roadGeoJsonUrl, '道路 GeoJSON')
}

export const loadRoadBoundary = (): Promise<FeatureCollection<Polygon | MultiPolygon>> => {
    return loadJson<FeatureCollection<Polygon | MultiPolygon>>(boundaryGeoJsonUrl, '行政区边界')
}

export {
    loadRasterDemoGrid,
    rasterDemoColorStops,
}

创建参数 ​

调用 createRoadLeafletLayer(options)，返回 Promise<RoadLeafletLayer>。

参数类型默认值必填说明
rendererType'webgl2' | 'webgpu'-是本页固定为 'webgl2'
roadsRoadGeoJsonInput-是LineString / MultiLineString GeoJSON
gridRasterGrid-二选一已加载规则栅格
sourceRasterSource-二选一TIF、MICAPS4 或 Params 数据源
colorStopsRasterColorStopInput[]-是非空色标数组
colorRangeRasterColorRange-否可见数值范围
colorMode'step' | 'smooth''step'否分段或平滑颜色
sampleMode'interpolate' | 'cell''interpolate'否双线性插值或原始格点
coordinateTransform'none' | 'wgs84-to-gcj02''none'否显示坐标转换
sampleCoordinateMode'source' | 'display''source'否栅格采样坐标来源
lineWidthnumber3否CSS 像素线宽，不能小于 0
opacitynumber1否0 到 1
panestringoverlayPane否Leaflet Pane
clipCoverageOptions-否多边形边界与裁剪模式

图层方法 ​

方法参数返回值说明
redraw-void请求重新绘制
setOpacitynumbervoid更新道路透明度
setLineWidthnumbervoid更新 CSS 像素线宽
setColorMode'step' | 'smooth'void更新颜色映射
setSampleMode'interpolate' | 'cell'void更新栅格采样
setClipModeClipModevoid复用已有边界切换裁剪方式

限制 ​

  • 浏览器必须支持 WebGL2。
  • 创建后不能通过公开图层方法替换道路、栅格、色标或 colorRange；这些输入变化时需要重建图层。
  • 图层移除时会销毁 WebGL2 资源，Vue 页面卸载时应调用 map.remove()。
  • 示例底图来自 OpenStreetMap，需要网络连接。
